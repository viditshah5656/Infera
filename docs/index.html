from openai import OpenAI client = OpenAI( base_url="http://127.0.0.1:8081/v1", api_key="local" ) reply = client.chat.completions.create( model="YOUR_MODEL", messages=[{"role":"user","content":"Hello"}] ) print(reply.choices[0].message.content)
